function messyProcess(a, b, c) {
  let r = 0;
  if (a) {
    for (let i = 0; i < b; i++) {
      while (c > 0) {
        if (a && b || c) {
          r += i;
        } else if (c) {
          r -= i;
        }
        c = c - 1;
      }
    }
  }
  if (a && b && c) {
    r = r * 2;
  }
  return r;
}

void messyProcess;
