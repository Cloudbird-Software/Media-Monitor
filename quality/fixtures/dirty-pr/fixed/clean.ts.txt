function normalizePart(value, bound) {
  if (value > bound) {
    return bound;
  }
  if (value < -bound) {
    return -bound;
  }
  return value;
}

function accumulate(total, next) {
  return total + next;
}

function cleanProcess(a, b, c) {
  const parts = [normalizePart(a, 100), normalizePart(b, 100), normalizePart(c, 100)];
  return parts.reduce(accumulate, 0);
}
